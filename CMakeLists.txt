cmake_minimum_required(VERSION 3.24)
if(CMAKE_HOST_APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS version")
endif()
project(RMX VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(RMX_BUILD_NATIVE "Build the native macOS application" ON)
option(RMX_BUILD_TESTS "Build engine checks" ON)
option(RMX_BUILD_PLUGINS "Build JUCE standalone, AU and VST3 adapters" OFF)
set(RMX_SANITIZER "none" CACHE STRING "Runtime checks: none, address, thread")
set_property(CACHE RMX_SANITIZER PROPERTY STRINGS none address thread)
if(RMX_SANITIZER STREQUAL "address")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(RMX_SANITIZER STREQUAL "thread")
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
elseif(NOT RMX_SANITIZER STREQUAL "none")
  message(FATAL_ERROR "RMX_SANITIZER must be none, address, or thread")
endif()
set(RMX_JUCE_PATH "" CACHE PATH "Existing JUCE 8.0.6 source checkout")

set(RMX_DSP_SOURCES
    src/core/Engine.cpp src/core/Primitives.cpp src/core/Sampler.cpp
    src/core/Isolator.cpp src/core/SceneEffects.cpp src/core/ReleaseEffects.cpp
    src/core/SampleBank.cpp)
add_library(rmx_core ${RMX_DSP_SOURCES} src/core/State.cpp src/core/Wav.cpp)
target_include_directories(rmx_core PUBLIC include)
set_target_properties(rmx_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(rmx_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(rmx_render tools/render.cpp)
target_link_libraries(rmx_render PRIVATE rmx_core)
add_executable(rmx_benchmark tools/benchmark.cpp)
target_link_libraries(rmx_benchmark PRIVATE rmx_core)
add_executable(rmx_hardware_profile tools/hardware_profile.cpp)
target_link_libraries(rmx_hardware_profile PRIVATE rmx_core)
if(RMX_BUILD_TESTS)
  enable_testing()
  add_executable(rmx_tests tests/core_tests.cpp)
  target_link_libraries(rmx_tests PRIVATE rmx_core)
  add_test(NAME engine COMMAND rmx_tests)
  foreach(case neutral_stereo sampler_panic echo_tail effect_extremes scene_processing realtime_allocations
               state_roundtrip queue_bounds panel_actions tempo_detection wav_roundtrip)
    add_test(NAME engine_${case} COMMAND rmx_tests ${case})
  endforeach()
  find_package(Threads REQUIRED)
  foreach(suite midi_controls application_session queue_concurrency)
    add_executable(rmx_${suite}_tests tests/${suite}_tests.cpp)
    target_link_libraries(rmx_${suite}_tests PRIVATE rmx_core Threads::Threads)
    add_test(NAME ${suite} COMMAND rmx_${suite}_tests)
  endforeach()
  add_executable(rmx_hardware_controls_tests tests/hardware_controls_tests.cpp)
  target_link_libraries(rmx_hardware_controls_tests PRIVATE rmx_core)
  add_test(NAME hardware_controls COMMAND rmx_hardware_controls_tests)
  foreach(suite audio_arena hardware_wire control_expanders connection_mode relay_feedback external_converter input_qualification)
    add_executable(rmx_${suite}_tests tests/${suite}_tests.cpp)
    add_test(NAME ${suite} COMMAND rmx_${suite}_tests)
  endforeach()
  target_compile_definitions(rmx_external_converter_tests PRIVATE RMX_EXTERNAL_CONVERTER=1)
  find_package(Python3 3.11 QUIET COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME electronics_tools COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p "test_electronics.py")
    set_tests_properties(electronics_tools PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
endif()
if(APPLE AND RMX_BUILD_NATIVE)
  enable_language(OBJCXX)
  add_library(rmx_macos src/macos/AudioHost.mm src/macos/ApplicationController.mm)
  target_link_libraries(rmx_macos PUBLIC rmx_core "-framework Cocoa" "-framework CoreAudio" "-framework CoreMIDI" "-framework AudioToolbox")
  target_compile_options(rmx_macos PRIVATE -fobjc-arc -Wall -Wextra -Wpedantic)
  add_executable(RMX MACOSX_BUNDLE src/macos/Main.mm)
  target_sources(RMX PRIVATE assets/rmx-1000_1.png)
  set_source_files_properties(assets/rmx-1000_1.png PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  target_link_libraries(RMX PRIVATE rmx_macos "-framework UniformTypeIdentifiers" "-framework AVFoundation")
  target_compile_options(RMX PRIVATE -fobjc-arc)
  set_target_properties(RMX PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/src/macos/Info.plist" MACOSX_BUNDLE_GUI_IDENTIFIER "nz.axelmckenna.rmx" MACOSX_BUNDLE_BUNDLE_NAME "RMX" MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}" XCODE_ATTRIBUTE_CODE_SIGN_IDENTITY "-")
  if(RMX_BUILD_TESTS)
    add_executable(rmx_application_controller_tests tests/application_controller_tests.mm)
    target_link_libraries(rmx_application_controller_tests PRIVATE rmx_macos)
    target_compile_options(rmx_application_controller_tests PRIVATE -fobjc-arc)
    add_test(NAME application_controller COMMAND rmx_application_controller_tests)
  endif()
endif()
if(RMX_BUILD_PLUGINS)
  if(RMX_JUCE_PATH)
    add_subdirectory("${RMX_JUCE_PATH}" "${CMAKE_BINARY_DIR}/JUCE")
  else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(JUCE)
  endif()
  set(formats VST3 Standalone)
  if(APPLE)
    list(APPEND formats AU)
  endif()
  juce_add_plugin(RMXPlugin COMPANY_NAME "Axel McKenna" BUNDLE_ID "nz.axelmckenna.rmx.plugin" PLUGIN_MANUFACTURER_CODE AxMc PLUGIN_CODE Rmxf FORMATS ${formats} PRODUCT_NAME "RMX" IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
  target_sources(RMXPlugin PRIVATE src/plugin/Plugin.cpp src/plugin/RMXProcessor.cpp)
  juce_add_binary_data(rmx_panel SOURCES assets/rmx-1000_1.png)
  target_link_libraries(RMXPlugin PRIVATE rmx_core rmx_panel juce::juce_audio_utils juce::juce_dsp)
  target_compile_definitions(RMXPlugin PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  if(RMX_BUILD_TESTS)
    add_executable(rmx_plugin_tests tests/plugin_tests.cpp)
    target_link_libraries(rmx_plugin_tests PRIVATE RMXPlugin)
    target_include_directories(rmx_plugin_tests PRIVATE $<TARGET_PROPERTY:RMXPlugin,INCLUDE_DIRECTORIES>)
    target_compile_definitions(rmx_plugin_tests PRIVATE $<TARGET_PROPERTY:RMXPlugin,COMPILE_DEFINITIONS>)
    add_test(NAME plugin_state COMMAND rmx_plugin_tests)
  endif()
endif()
