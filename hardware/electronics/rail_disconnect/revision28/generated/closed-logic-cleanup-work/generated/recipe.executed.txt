"""Remove only natively dangling ordinary objects on already-closed logic nets.

Force, timing, reference, power, every open net and all latch-incident nets stay
fixed. Fresh native checks must prove that actual package pads stay connected.
"""

import hashlib
import json
import shutil
from pathlib import Path

from ...common.sexpr import children, first, read_sexpr, sexpr
from .ordinary_joint_ripup7 import ORDINARY
from .schmitt_cone_candidate import GROUPS

ROOT = Path(__file__).resolve().parent
SOURCE = ROOT / "generated/pg-master-local-repair-work"
WORK = ROOT / "generated/closed-logic-cleanup-work"


def main():
    assert not WORK.exists()
    source = SOURCE / "rail_disconnect.kicad_pcb"
    blob = source.read_bytes()
    digest = hashlib.sha256(blob).hexdigest()
    assert digest == "09265b396dbb002eaa8127d569d69d6aa93291ea79db57c137fcf3fe5073d05b"
    tree = read_sexpr(blob.decode())
    names = {str(n[1]): str(n[2]).lstrip("/") for n in children(tree, "net")}
    objects = {str(first(n, "uuid")[1]): n for k in ("segment", "via") for n in children(tree, k)}
    parts = json.loads((SOURCE / "components.json").read_text())
    protected = {p["net"] for c in parts if c["ref"] in ("U21", "U22") for p in c["pins"]}
    graph_path = SOURCE / "generated/residual-components.json"
    graph = json.loads(graph_path.read_text())
    assert graph["source_sha256"][str(source.resolve())] == digest
    signals = {n for group in GROUPS for n in group if n}
    allowed = (ORDINARY | signals | {n + "_SCHMITT" for n in signals}) - protected - set(graph["inventory"])
    raw = SOURCE / "generated/drc-pg-master-local-repair.json"
    drc = json.loads(raw.read_text())
    assert not drc["unconnected_items"] == []
    removed = []
    seen = set()
    for violation in drc["violations"]:
        if violation["type"] not in ("track_dangling", "via_dangling"):
            continue
        for item in violation["items"]:
            uid = item["uuid"]
            node = objects.get(uid)
            if node is None or uid in seen:
                continue
            net = names[str(first(node, "net")[1])]
            if net not in allowed:
                continue
            seen.add(uid)
            removed.append(dict(uuid=uid, net=net, exact_original_object=sexpr(node), raw_native_finding=violation))
            tree.remove(node)
    assert len(removed) == 48
    WORK.mkdir()
    (WORK / "generated").mkdir()
    for path in SOURCE.iterdir():
        if path.is_file() and (path.suffix in (".kicad_pcb", ".kicad_sch", ".kicad_pro", ".kicad_sym") or path.name in ("components.json", "fp-lib-table", "sym-lib-table")):
            shutil.copy2(path, WORK / path.name)
        elif path.suffix == ".pretty":
            shutil.copytree(path, WORK / path.name)
    (WORK / source.name).write_text(sexpr(tree) + "\n")
    report = dict(source_pcb_sha256=digest, native_graph_sha256=hashlib.sha256(graph_path.read_bytes()).hexdigest(), raw_DRC_sha256=hashlib.sha256(raw.read_bytes()).hexdigest(), recipe_sha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest(), protected_latch_incident_nets=sorted(n for n in protected if n), removed_ordinary_dangling_objects=removed, actual_pad_connectivity_must_remain_unchanged=True, all_force_timing_reference_power_open_nets_and_part_geometry_fixed=True, fresh_native_and_finite_copper_checks_required=True, physical_qualified=False)
    (WORK / "generated/closed-logic-cleanup.json").write_text(json.dumps(report, indent=2) + "\n")
    assert source.read_bytes() == blob
    print("Removed48 literal dangling ordinary objects on closed nets", flush=True)


if __name__ == "__main__":
    main()
