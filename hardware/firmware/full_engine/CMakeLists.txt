cmake_minimum_required(VERSION 3.24)
project(RMXFullEngineFirmware LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(RMX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_library(rmx_full_core
  ${RMX_ROOT}/src/core/drums/Kick.cpp ${RMX_ROOT}/src/core/drums/Snare.cpp
  ${RMX_ROOT}/src/core/drums/Clap.cpp ${RMX_ROOT}/src/core/drums/Hat.cpp
  ${RMX_ROOT}/src/core/Engine.cpp ${RMX_ROOT}/src/core/Scene.cpp
  ${RMX_ROOT}/src/core/Isolator.cpp ${RMX_ROOT}/src/core/CardSamples.cpp
  ${RMX_ROOT}/src/core/CardSettings.cpp ${RMX_ROOT}/src/core/State.cpp
  ${RMX_ROOT}/src/core/Wav.cpp ${RMX_ROOT}/src/core/CardExport.cpp BootStorage.cpp KernelTransport.cpp)
set_source_files_properties(${RMX_ROOT}/src/core/drums/Kick.cpp ${RMX_ROOT}/src/core/drums/Snare.cpp ${RMX_ROOT}/src/core/drums/Clap.cpp ${RMX_ROOT}/src/core/drums/Hat.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
target_compile_definitions(rmx_full_core PUBLIC RMX_EMBEDDED_COMPRESSED_KERNELS=1)
target_include_directories(rmx_full_core PUBLIC include ${RMX_ROOT}/include ${RMX_ROOT})
target_compile_options(rmx_full_core PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(full_engine ${RMX_ROOT}/tests/core_tests.cpp tests/bootstrap.cpp)
target_link_libraries(full_engine PRIVATE rmx_full_core)
add_test(NAME full_engine COMMAND full_engine)
set_tests_properties(full_engine PROPERTIES WORKING_DIRECTORY ${RMX_ROOT})
foreach(suite tempo parity isolator scene sampler xpad card card_settings hardware_calibration)
  add_executable(full_${suite} ${RMX_ROOT}/tests/${suite}_tests.cpp tests/bootstrap.cpp)
  target_link_libraries(full_${suite} PRIVATE rmx_full_core)
  add_test(NAME ${suite} COMMAND full_${suite})
  set_tests_properties(${suite} PROPERTIES WORKING_DIRECTORY ${RMX_ROOT})
endforeach()
add_executable(full_memory ${RMX_ROOT}/tools/memory_budget.cpp tests/bootstrap.cpp)
target_link_libraries(full_memory PRIVATE rmx_full_core)
add_test(NAME full_memory COMMAND full_memory)
add_executable(full_storage tests/storage_fault_tests.cpp tests/bootstrap.cpp)
target_include_directories(full_storage PRIVATE ${RMX_ROOT}/hardware/runtime/tests/support)
target_link_libraries(full_storage PRIVATE rmx_full_core)
add_test(NAME full_storage COMMAND full_storage)
add_executable(full_integration tests/integration_tests.cpp tests/bootstrap.cpp)
target_link_libraries(full_integration PRIVATE rmx_full_core)
add_test(NAME full_integration COMMAND full_integration)
add_executable(full_kernels tests/kernel_tests.cpp)
target_link_libraries(full_kernels PRIVATE rmx_full_core)
add_test(NAME full_kernels COMMAND full_kernels)
add_executable(full_control_recovery tests/control_recovery_tests.cpp)
add_test(NAME full_control_recovery COMMAND full_control_recovery)
add_executable(full_allocator_storage tests/allocator_storage_tests.cpp)
target_link_libraries(full_allocator_storage PRIVATE rmx_full_core)
add_test(NAME full_allocator_storage COMMAND full_allocator_storage)
