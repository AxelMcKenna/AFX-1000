// TBH05 spring attachment trial. Existing bracket/arm eye holes unchanged.
include <carrier.scad>
// All shaft axes X. Brass bolts/sleeves reduce nearby ferromagnetic material.
// Full spring loops slide onto a removable smooth sleeve, not through eye holes.
module spring_pin(y,z){
 translate([311.4,y,z])rotate([0,90,0])cylinder(d=3.8,h=1.6);
 translate([313,y,z])rotate([0,90,0])cylinder(d=2,h=12);
 translate([319,y,z])rotate([0,90,0])difference(){cylinder(d=4,h=2);translate([0,0,-.01])cylinder(d=2.3,h=2.02);}
 translate([319,y,z])rotate([0,90,0])difference(){cylinder(d=2.2,h=2.7);translate([0,0,-.01])cylinder(d=2.1,h=2.72);}
 translate([321.7,y,z])rotate([0,90,0])difference(){cylinder(d=4.5,h=.4);translate([0,0,-.01])cylinder(d=2.1,h=.42);}
 translate([322.1,y,z])rotate([0,90,0])difference(){cylinder(d=4.7,h=2.5,$fn=6);translate([0,0,-.01])cylinder(d=1.6,h=2.52);}
}
module fixed_pin(){spring_pin(70,35.6);}
module moving_pin(a=0){translate([0,82,31.6])rotate([-a,0,0])translate([0,-82,-31.6])spring_pin(82,41.6);}
// Full connected spring occupied volume, including loop and coil maxima.
// Spring center X321.35 with ±0.17 axial loop play and 0.10 radial allowance.
module attached_spring_corridor(){for(a=[0:1:34])hull(){
 for(dx=[-.17,.17])translate([321.35+dx,70,35.6])sphere(d=3.38,$fn=32);
 for(t=[a,a+1],dx=[-.17,.17])translate([321.35+dx,82+10*sin(t),31.6+10*cos(t)])sphere(d=3.38,$fn=32);
}}
module pin_sweep(){for(a=[0:1:34])hull(){moving_pin(a);moving_pin(a+1);}}
module spring_installed_envelope(){fixed_pin();pin_sweep();attached_spring_corridor();}
if(part=="spring_hardware_controls_clearance")intersection(){spring_installed_envelope();union(){translate([0,0,H])lid();guide_installed();holders_installed();}}
if(part=="spring_hardware_shell_clearance")intersection(){spring_installed_envelope();shell();}
if(part=="spring_hardware_components_clearance")intersection(){spring_installed_envelope();box_rows(component_boxes);}
if(part=="spring_hardware_carrier_clearance")intersection(){spring_installed_envelope();tbh05_installed();}
// Sleeve/hardware intentionally engage existing eyes; spring body excludes ends.
module spring_body_sweep(){for(a=[0:1:34])hull()for(t=[a,a+1]){
 fy=70;fz=35.6;my=82+10*sin(t);mz=31.6+10*cos(t);l=sqrt(pow(my-fy,2)+pow(mz-fz,2));
 for(f=[3/l,1-3/l],dx=[-.17,.17])translate([321.35+dx,fy+(my-fy)*f,fz+(mz-fz)*f])sphere(d=3.38,$fn=32);
}}
if(part=="spring_body_lever_clearance")intersection(){spring_body_sweep();union(){bracket_world();guard();axial_motion(.5);}}
